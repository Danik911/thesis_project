## Objective

Add Snowflake as a second data source alongside file upload. Users select a "Snowflake" tab on the idle page, enter connection credentials, browse available tables or stage files, and load data into an in-memory BI session. All downstream features (filters, copilot, charts, export) remain unchanged — they operate on `pd.DataFrame` regardless of origin.

**Key architectural insight:** Everything after `session_store.create_session()` is data-source agnostic. We only need a new path: Snowflake → `pd.DataFrame` → `create_session()`.

**Authentication:** Username/password for personal account first. Key-pair auth for company accounts can be added later without breaking changes.

**Package:** `snowflake-connector-python[pandas]` — uses `cursor.fetch_pandas_all()` to return DataFrames directly. No SQLAlchemy needed (native connector handles stage operations which SQLAlchemy cannot).

### 1. snowflake_connector.py — Backend Connector

```python
"""Snowflake data source connector for MES Agentic BI."""

from __future__ import annotations

import logging
import pathlib
import tempfile
from typing import Any

import pandas as pd

from .config import get_bi_config

logger = logging.getLogger(__name__)


def _get_connection(
    account: str,
    user: str,
    password: str,
    warehouse: str,
    database: str,
    schema: str,
) -> Any:
    """Open a Snowflake connection. Raises on failure — no fallbacks."""
    try:
        import snowflake.connector
    except ImportError as exc:
        raise RuntimeError(
            "snowflake-connector-python is not installed. "
            "Run: uv add 'snowflake-connector-python[pandas]'"
        ) from exc

    return snowflake.connector.connect(
        account=account,
        user=user,
        password=password,
        warehouse=warehouse,
        database=database,
        schema=schema,
        client_session_keep_alive=False,  # per-request connections for PoC
    )


def list_tables(
    account: str,
    user: str,
    password: str,
    warehouse: str,
    database: str,
    schema: str,
) -> list[dict[str, Any]]:
    """Return list of tables/views in the given schema.

    Each item: {"name": str, "kind": "BASE TABLE"|"VIEW", "row_count": int|None}
    Raises on authentication or query failure.
    """
    conn = _get_connection(account, user, password, warehouse, database, schema)
    try:
        cur = conn.cursor()
        cur.execute(
            "SELECT TABLE_NAME, TABLE_TYPE, ROW_COUNT "
            "FROM INFORMATION_SCHEMA.TABLES "
            "WHERE TABLE_SCHEMA = %s "
            "ORDER BY TABLE_NAME",
            (schema.upper(),),
        )
        rows = cur.fetchall()
        return [
            {
                "name": row[0],
                "kind": row[1],
                "row_count": int(row[2]) if row[2] is not None else None,
            }
            for row in rows
        ]
    finally:
        conn.close()


def list_stage_files(
    account: str,
    user: str,
    password: str,
    warehouse: str,
    database: str,
    schema: str,
    stage_name: str,
) -> list[dict[str, Any]]:
    """Return list of files in a named Snowflake stage.

    Each item: {"name": str, "size": int, "last_modified": str}
    Raises on authentication or query failure.
    """
    conn = _get_connection(account, user, password, warehouse, database, schema)
    try:
        cur = conn.cursor()
        cur.execute(f"LIST @{stage_name}")
        rows = cur.fetchall()
        # LIST @stage returns: (name, size, md5, last_modified)
        return [
            {
                "name": row[0],
                "size": int(row[1]) if row[1] else 0,
                "last_modified": str(row[3]) if len(row) > 3 else "",
            }
            for row in rows
        ]
    finally:
        conn.close()


def fetch_table(
    account: str,
    user: str,
    password: str,
    warehouse: str,
    database: str,
    schema: str,
    table_name: str,
    limit: int | None = None,
) -> pd.DataFrame:
    """Fetch a Snowflake table/view as a pandas DataFrame.

    Args:
        limit: Row cap. Defaults to BI_MAX_ROWS from config.

    Raises:
        RuntimeError: If snowflake-connector-python not installed.
        ValueError: If DataFrame is empty or exceeds max_rows.
        snowflake.connector errors: On auth/query failure.
    """
    config = get_bi_config()
    effective_limit = limit if limit is not None else config.max_rows

    conn = _get_connection(account, user, password, warehouse, database, schema)
    try:
        cur = conn.cursor()
        # Quote table name to handle case sensitivity
        sql = f'SELECT * FROM "{table_name}" LIMIT {effective_limit}'
        cur.execute(sql)
        df = cur.fetch_pandas_all()
    finally:
        conn.close()

    df = df.convert_dtypes()

    if df.empty:
        raise ValueError(f"Snowflake table '{table_name}' returned no rows")

    if len(df) > config.max_rows:
        raise ValueError(
            f"Snowflake table '{table_name}' has {len(df)} rows, "
            f"exceeds BI_MAX_ROWS={config.max_rows}"
        )

    logger.info(
        "fetch_table: account=%s table=%s rows=%d cols=%d",
        account, table_name, len(df), len(df.columns),
    )
    return df


def fetch_stage_file(
    account: str,
    user: str,
    password: str,
    warehouse: str,
    database: str,
    schema: str,
    stage_name: str,
    file_path: str,
) -> pd.DataFrame:
    """Download a file from a Snowflake stage and parse as DataFrame.

    Supports .xlsx and .csv stage files. Downloads via GET command to
    temp directory, then parses with existing data_parser.parse_file().

    Raises:
        RuntimeError: If snowflake-connector-python not installed.
        ValueError: If file type unsupported, empty, or exceeds max_rows.
        snowflake.connector errors: On auth/GET failure.
    """
    from .data_parser import parse_file  # avoid circular import

    conn = _get_connection(account, user, password, warehouse, database, schema)
    try:
        with tempfile.TemporaryDirectory() as tmp_dir:
            cur = conn.cursor()
            cur.execute(f"GET @{stage_name}/{file_path} file://{tmp_dir}/")

            tmp_path = pathlib.Path(tmp_dir)
            downloaded = list(tmp_path.iterdir())
            if not downloaded:
                raise ValueError(
                    f"GET @{stage_name}/{file_path} produced no output file"
                )

            local_file = downloaded[0]
            content = local_file.read_bytes()
            # Use original file_path name for extension detection
            filename = pathlib.Path(file_path).name
            df = parse_file(content, filename)
    finally:
        conn.close()

    logger.info(
        "fetch_stage_file: account=%s stage=%s file=%s rows=%d cols=%d",
        account, stage_name, file_path, len(df), len(df.columns),
    )
    return df
```

@router.post("/snowflake/tables")

        tables = sf_list_tables(
            account=request.account, user=request.user,
            password=request.password, warehouse=request.warehouse,
            database=request.database, schema=request.schema_name,

        return {"tables": tables, "database": request.database, "schema": request.schema_name}

            database=request.database, schema=request.schema_name,

        "session_id": session.session_id,
        "filename": session.filename,
        "total_rows": session.total_rows,
        "total_columns": session.total_columns,
        "columns": [column.model_dump() for column in session.columns],

## Design Decisions

| Decision | Choice | Rationale |
|----------|--------|-----------|
| **All endpoints POST** | Credentials in JSON body | Never expose passwords in URLs (logged by proxies, browser history) |
| **Per-request connections** | `_get_connection()` opens/closes each call | Simplest for PoC. No connection pool complexity. ~300ms overhead per request is acceptable. |
| **`INFORMATION_SCHEMA.TABLES`** | Not `SHOW TABLES` | Returns structured SQL result set compatible with `fetchall()`. Predictable column names. |
| **Native connector** | Not SQLAlchemy | Lighter dependency. Supports stage operations (GET/LIST) which SQLAlchemy cannot do. |
| **`fetch_pandas_all()`** | Not `fetch_pandas_batches()` | BI sessions cap at `BI_MAX_ROWS` (100K). Single DataFrame fits in memory. |
| **Separate component** | `SnowflakeBrowser.tsx` | Keeps `agentic-bi.tsx` manageable. Same pattern as `ChatDrawer`, `Sidebar`. |
| **Load returns `BIUploadResponse` shape** | Same response as `/upload` | Frontend reuses identical state-setting logic. Zero changes to grid, filters, copilot, export. |
| **`tempfile.TemporaryDirectory`** | For stage file GET | Auto-cleanup on success or failure. Snowflake GET requires local filesystem path. |
