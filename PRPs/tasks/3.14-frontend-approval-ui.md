```tsx
import { useState, useEffect } from 'react';
import { useAuth, useUser } from '@clerk/nextjs';
import { Dialog } from '@headlessui/react';

interface CategorizationResult {
  category: number;
  confidence_score: number;
  reasoning: string;
  has_ambiguity_signals: boolean;
  ambiguity_details: string;
  alternative_categories: number[];
}

interface ApprovalModalProps {
  isOpen: boolean;
  onClose: () => void;
  jobId: string;
  categorizationResult: CategorizationResult;
  onApprovalSubmitted: () => void;
}

export default function ApprovalModal({
  isOpen,
  onClose,
  jobId,
  categorizationResult,
  onApprovalSubmitted
}: ApprovalModalProps) {
  const { getToken } = useAuth();
  const { user } = useUser();

  const [approvalDecision, setApprovalDecision] = useState<'APPROVE' | 'REJECT' | 'REQUEST_REVISION'>('APPROVE');
  const [humanCategory, setHumanCategory] = useState<number>(categorizationResult.category);
  const [justification, setJustification] = useState('');
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  // Generate digital signature
  const generateSignature = () => {
    const timestamp = new Date().toISOString().replace(/[:.]/g, '').slice(0, 15);
    return `${user?.id}_${timestamp}`;
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);

    // Validation
    if (justification.length < 10) {
      setError('Justification must be at least 10 characters for audit trail compliance.');
      return;
    }

    setIsSubmitting(true);

    try {
      const token = await getToken();

      const response = await fetch(`/api/jobs/${jobId}/approval`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${token}`
        },
        body: JSON.stringify({
          approval_decision: approvalDecision,
          human_category: approvalDecision === 'APPROVE' ? humanCategory : null,
          justification,
          user_signature: generateSignature()
        })
      });

      if (!response.ok) {
        const errorData = await response.json();
        throw new Error(errorData.detail || `HTTP ${response.status}: ${response.statusText}`);
      }

      const result = await response.json();
      console.log('Approval submitted successfully:', result);

      // Notify parent component
      onApprovalSubmitted();
      onClose();

    } catch (err: any) {
      console.error('Error submitting approval:', err);
      setError(err.message || 'Failed to submit approval decision');
    } finally {
      setIsSubmitting(false);
    }
  };

  // Color coding for confidence score
  const getConfidenceColor = (score: number) => {
    if (score < 0.7) return 'text-red-600 bg-red-50';
    if (score < 0.85) return 'text-yellow-600 bg-yellow-50';
    return 'text-green-600 bg-green-50';
  };

  return (
    <Dialog open={isOpen} onClose={onClose} className="relative z-50">
      {/* Backdrop */}
      <div className="fixed inset-0 bg-black/50" aria-hidden="true" />

      {/* Modal */}
      <div className="fixed inset-0 flex items-center justify-center p-4">
        <Dialog.Panel className="mx-auto max-w-3xl w-full rounded-lg bg-white shadow-xl">
          {/* Header */}
          <div className="bg-blue-600 text-white px-6 py-4 rounded-t-lg">
            <Dialog.Title className="text-xl font-bold">
              🤖 AI Categorization Review Required
            </Dialog.Title>
            <p className="text-sm text-blue-100 mt-1">
              The AI detected ambiguity and requires human oversight per GAMP-5 and EU AI Act Article 14.
            </p>
          </div>

          <form onSubmit={handleSubmit} className="px-6 py-4">
            {/* AI Recommendation Section */}
            <div className="mb-6 p-4 bg-gray-50 rounded border border-gray-200">
              <h3 className="text-lg font-semibold text-gray-900 mb-3">
                AI Recommendation
              </h3>

              {/* Category + Confidence */}
              <div className="grid grid-cols-2 gap-4 mb-3">
                <div>
                  <p className="text-sm text-gray-600 mb-1">Suggested Category</p>
                  <p className="text-2xl font-bold text-blue-600">
                    Category {categorizationResult.category}
                  </p>
                </div>
                <div>
                  <p className="text-sm text-gray-600 mb-1">Confidence Score</p>
                  <p className={`text-2xl font-bold px-3 py-1 rounded inline-block ${getConfidenceColor(categorizationResult.confidence_score)}`}>
                    {(categorizationResult.confidence_score * 100).toFixed(0)}%
                  </p>
                </div>
              </div>

              {/* Ambiguity Alert */}
              {categorizationResult.has_ambiguity_signals && (
                <div className="mb-3 p-3 bg-yellow-50 border border-yellow-300 rounded">
                  <p className="text-sm font-semibold text-yellow-800 mb-1">
                    ⚠️ Ambiguity Detected
                  </p>
                  <p className="text-sm text-yellow-700">
                    {categorizationResult.ambiguity_details}
                  </p>
                </div>
              )}

              {/* Alternative Categories */}
              {categorizationResult.alternative_categories && categorizationResult.alternative_categories.length > 0 && (
                <div className="mb-3">
                  <p className="text-sm text-gray-600 mb-1">Alternative Categories</p>
                  <div className="flex gap-2">
                    {categorizationResult.alternative_categories.map(cat => (
                      <span key={cat} className="px-3 py-1 bg-blue-100 text-blue-700 rounded text-sm font-medium">
                        Category {cat}
                      </span>
                    ))}
                  </div>
                </div>
              )}

              {/* AI Reasoning */}
              <div>
                <p className="text-sm text-gray-600 mb-1">AI Reasoning</p>
                <p className="text-sm text-gray-700 bg-white p-3 rounded border border-gray-200">
                  {categorizationResult.reasoning}
                </p>
              </div>
            </div>

            {/* Human Decision Section */}
            <div className="mb-6">
              <h3 className="text-lg font-semibold text-gray-900 mb-3">
                Your Decision
              </h3>

              {/* Decision Select */}
              <div className="mb-4">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Approval Decision <span className="text-red-500">*</span>
                </label>
                <select
                  value={approvalDecision}
                  onChange={(e) => setApprovalDecision(e.target.value as any)}
                  className="w-full px-3 py-2 border border-gray-300 rounded focus:ring-2 focus:ring-blue-500"
                  required
                >
                  <option value="APPROVE">✅ Approve (accept AI recommendation or override)</option>
                  <option value="REJECT">❌ Reject (cancel test generation)</option>
                  <option value="REQUEST_REVISION">🔄 Request Revision (needs more analysis)</option>
                </select>
              </div>

              {/* Category Override (only for APPROVE) */}
              {approvalDecision === 'APPROVE' && (
                <div className="mb-4">
                  <label className="block text-sm font-medium text-gray-700 mb-2">
                    Final GAMP-5 Category <span className="text-red-500">*</span>
                  </label>
                  <select
                    value={humanCategory}
                    onChange={(e) => setHumanCategory(Number(e.target.value))}
                    className="w-full px-3 py-2 border border-gray-300 rounded focus:ring-2 focus:ring-blue-500"
                    required
                  >
                    <option value={1}>Category 1 - Infrastructure (OS, firmware)</option>
                    <option value={3}>Category 3 - Non-configured COTS</option>
                    <option value={4}>Category 4 - Configured COTS</option>
                    <option value={5}>Category 5 - Custom-developed</option>
                  </select>
                  <p className="text-xs text-gray-500 mt-1">
                    You can override the AI recommendation if you disagree.

                </div>

              <div className="mb-4">
                <label className="block text-sm font-medium text-gray-700 mb-2">
                  Justification <span className="text-red-500">*</span>

                <textarea
                  value={justification}
                  onChange={(e) => setJustification(e.target.value)}
                  className="w-full px-3 py-2 border border-gray-300 rounded focus:ring-2 focus:ring-blue-500"
                  rows={4}
                  placeholder="Explain your decision (minimum 10 characters for audit trail compliance)"
                  required
                  minLength={10}
                />
                <p className="text-xs text-gray-500 mt-1">
                  {justification.length}/10 characters minimum (ALCOA+ requirement)
                </p>
              </div>

              {/* Digital Signature Info */}
              <div className="mb-4 p-3 bg-blue-50 border border-blue-200 rounded">
                <p className="text-xs text-blue-700">
                  <strong>Digital Signature:</strong> {user?.id}_[timestamp]
                </p>
                <p className="text-xs text-blue-600 mt-1">
                  Your decision will be recorded with your user ID, email, and timestamp per 21 CFR Part 11 §11.50.
                </p>
              </div>
            </div>

            {/* Error Display */}
            {error && (
              <div className="mb-4 p-3 bg-red-50 border border-red-300 rounded">
                <p className="text-sm text-red-800">
                  <strong>Error:</strong> {error}
                </p>
              </div>
            )}

            {/* Footer Buttons */}
            <div className="flex justify-end gap-3 pt-4 border-t border-gray-200">
              <button
                type="button"
                onClick={onClose}
                className="px-4 py-2 text-gray-700 bg-gray-100 rounded hover:bg-gray-200"
                disabled={isSubmitting}
              >
                Cancel
              </button>

                type="submit"
                className="px-6 py-2 bg-blue-600 text-white rounded hover:bg-blue-700 disabled:bg-gray-400 disabled:cursor-not-allowed"
                disabled={isSubmitting || justification.length < 10}
              >
                {isSubmitting ? 'Submitting...' : 'Submit Decision'}
              </button>
            </div>
          </form>
        </Dialog.Panel>
      </div>
    </Dialog>
  );
}
```

### Phase 2: Add Job Status Polling Hook (1 hour)

**File**: `main/frontend/hooks/useJobStatusPolling.ts` (NEW)

**Custom Hook**:

```typescript
import { useState, useEffect, useCallback } from 'react';
import { useAuth } from '@clerk/nextjs';

interface JobStatus {
  job_id: string;
  status: 'pending' | 'processing' | 'awaiting_approval' | 'approved' | 'rejected' | 'completed' | 'failed';
  requires_approval: boolean;
  approval_reason: string | null;
  timeout_remaining_seconds: number | null;
  categorization_result: any;
  alternative_categories: number[] | null;
  created_at: string;
  updated_at: string;
}

export function useJobStatusPolling(jobId: string | null, intervalMs: number = 5000) {
  const { getToken } = useAuth();
  const [status, setStatus] = useState<JobStatus | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const fetchStatus = useCallback(async () => {
    if (!jobId) {
      setLoading(false);
      return;
    }

    try {
      const token = await getToken();

      const response = await fetch(`/api/jobs/${jobId}/status`, {
        headers: {
          'Authorization': `Bearer ${token}`
        }
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      const data = await response.json();
      setStatus(data);
      setError(null);
      setLoading(false);

    } catch (err: any) {
      console.error('Error fetching job status:', err);
      setError(err.message);
      setLoading(false);
    }
  }, [jobId, getToken]);

  useEffect(() => {
    // Initial fetch
    fetchStatus();

    // Poll every intervalMs
    const interval = setInterval(fetchStatus, intervalMs);

    // Cleanup
    return () => clearInterval(interval);
  }, [fetchStatus, intervalMs]);

  return { status, loading, error, refetch: fetchStatus };
}
```

**Usage**:
```tsx

  // Show ApprovalModal
}
```

```tsx
    case 'completed': return '✅ Completed';
    case 'rejected': return '❌ Rejected';
    case 'failed': return '❌ Failed';
```

**Key Frontend Patterns from Examples:**

1. **Status Polling Pattern** (from demo):
```typescript
// Poll job status to detect approval required
const pollJobStatus = async () => {
  const response = await fetch(`/api/jobs/${jobId}/status`);
  const data = await response.json();

  if (data.status === 'awaiting_approval') {
    setShowApprovalModal(true);
    setCategorizationResult(data.categorization_result);
  }
};

useEffect(() => {
  const interval = setInterval(pollJobStatus, 5000);
  return () => clearInterval(interval);
}, [jobId]);
```

2. **WebSocket Alternative** (from demo - production recommended):
```typescript
// Real-time status updates via WebSocket
const ws = new WebSocket(`wss://api/jobs/${jobId}/ws`);

ws.onmessage = (event) => {
  const data = JSON.parse(event.data);
  if (data.type === 'approval_required') {
    setShowApprovalModal(true);
  }
};
```

3. **Approval Form Submission**:
```typescript
const submitApproval = async (decision: ApprovalDecision) => {
  const response = await fetch(`/api/jobs/${jobId}/approval`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      approval_decision: decision.decision,
      human_category: decision.category,
      justification: decision.justification,
      user_signature: `${userId}_${timestamp}`
    })
  });

  if (response.ok) {
    setShowApprovalModal(false);
    // Workflow resumes automatically
  }
};
```
