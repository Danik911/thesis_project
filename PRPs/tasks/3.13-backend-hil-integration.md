## Objective
Implement backend API infrastructure to pause test generation workflows when `requires_human_review=true`, expose approval endpoints, and resume workflows after human decisions with complete ALCOA+ audit trail.

- ❌ API doesn't expose HIL events to frontend
- ❌ No job status state for "awaiting approval"
- ❌ No mechanism to pause/resume workflows via API
- ❌ No audit trail storage for human decisions

**Requirements**:
- Workflow must pause when categorization triggers `requires_human_review=true`
- Frontend must poll for approval-required status
- Human must approve/reject/revise categorization via API
- Workflow must resume with human decision
- All interactions must be ALCOA+ compliant (timestamped, attributed, signed)

import asyncio

@router.post(
    "/jobs/{job_id}/approval",
    response_model=ApprovalResponse,
    tags=["approval"],
    summary="Submit human approval decision",
    description="Resume paused workflow with human decision on GAMP-5 categorization"
)
async def submit_approval_decision(
    job_id: str,
    request: ApprovalRequest,
    claims: ClerkClaims = Depends(verify_token),
    db: Session = Depends(get_db),
    req: Request = None  # For extracting IP, user agent
):
    """
    Submit human approval decision for ambiguous categorization.

    Workflow:
    1. Validate job exists and requires approval
    2. Verify user authorization
    3. Store approval decision in audit trail
    4. Emit HumanResponseEvent to resume workflow
    5. Update job status to APPROVED/REJECTED
    6. Return updated status
    """

    # 1. Fetch job
    job = db.query(Job).filter_by(id=job_id).first()
    if not job:
        raise HTTPException(404, f"Job {job_id} not found")

    # 2. Verify ownership
    if job.user_id != claims.user_id:
        raise HTTPException(403, "Not authorized to approve this job")

    # 3. Validate job is awaiting approval
    if job.status != JobStatus.AWAITING_APPROVAL:
        raise HTTPException(
            400,
            f"Job status is '{job.status}', not awaiting approval"
        )

    # 4. Check timeout
    if job.approval_timeout_at and datetime.utcnow() > job.approval_timeout_at:
        job.status = JobStatus.REJECTED
        db.commit()
        raise HTTPException(
            408,
            "Approval timeout expired. Job automatically rejected."
        )

    # 5. Extract categorization details from job context
    categorization_data = job.context.get("categorization_result", {})
    ai_category = categorization_data.get("category")
    ai_confidence = categorization_data.get("confidence_score")
    ai_reasoning = categorization_data.get("reasoning", "")
    ambiguity_reason = categorization_data.get("ambiguity_details", "")
    alternative_categories = categorization_data.get("alternative_categories", [])

    # 6. Determine final category
    if request.approval_decision == ApprovalDecision.APPROVE:
        final_category = request.human_category or ai_category
    else:
        final_category = None  # Rejected, no category assigned

    # 7. Create approval record (ALCOA+ audit trail)
    approval_record = ApprovalDecisionRecord(
        job_id=job_id,
        # AI recommendation
        ai_category=ai_category,
        ai_confidence=ai_confidence,
        ai_reasoning=ai_reasoning,
        ambiguity_reason=ambiguity_reason,
        alternative_categories=alternative_categories,
        # Human decision
        approval_decision=request.approval_decision.value,
        human_category=final_category,
        justification=request.justification,
        # ALCOA+ compliance
        user_id=claims.user_id,
        user_email=claims.email,
        user_role=claims.metadata.get("role", "User"),
        digital_signature=request.user_signature,
        timestamp=datetime.utcnow(),
        # Metadata
        ip_address=req.client.host if req else None,
        user_agent=req.headers.get("User-Agent") if req else None
    )
    db.add(approval_record)

    # 8. Update job status
    if request.approval_decision == ApprovalDecision.APPROVE:
        job.status = JobStatus.APPROVED
        job.context["final_category"] = final_category
        job.context["approved_by"] = claims.user_id
        job.context["approved_at"] = datetime.utcnow().isoformat()
    elif request.approval_decision == ApprovalDecision.REJECT:
        job.status = JobStatus.REJECTED
    # REQUEST_REVISION handled in Phase 4 (resume with feedback)

    db.commit()

    # 9. Resume workflow by emitting HumanResponseEvent
    workflow_resumed = False
    if request.approval_decision == ApprovalDecision.APPROVE:
        try:
            # Emit event to workflow queue
            human_response = HumanResponseEvent(
                consultation_type="gamp5_categorization",
                decision=final_category,
                justification=request.justification,
                metadata={
                    "user_id": claims.user_id,
                    "user_email": claims.email,
                    "approval_decision": request.approval_decision.value,
                    "digital_signature": request.user_signature
                }
            )

            # Send to workflow (implementation depends on workflow queue mechanism)
            await resume_workflow(job_id, human_response)
            workflow_resumed = True

        except Exception as e:
            logger.error(f"Failed to resume workflow for job {job_id}: {e}")
            # Don't fail the approval - job is approved, workflow will retry

    # 10. Return response
    return ApprovalResponse(
        job_id=job_id,
        status=job.status,
        gamp_category=final_category or ai_category,
        workflow_resumed=workflow_resumed,
        trace_id=job.context.get("trace_id"),
        message=f"Approval decision '{request.approval_decision.value}' recorded successfully."
    )

from main.api.models import JobStatus
from main.src.core.events import HumanConsultationRequiredEvent, HumanResponseEvent
from datetime import datetime, timedelta

async def execute_workflow(job_id: str, db_session: Session):
    """
    Execute workflow with HIL pause/resume support.
    """

    # Fetch job
    job = db_session.query(Job).filter_by(id=job_id).first()
    if not job:
        logger.error(f"Job {job_id} not found")
        return

    try:
        # Update status to processing
        job.status = JobStatus.PROCESSING
        db_session.commit()

        # Initialize workflow
        workflow = UnifiedWorkflow(timeout=3600, verbose=True)

        # Prepare context with database session
        ctx = workflow.create_context(
            job_id=job_id,
            urs_content=job.urs_content,
            db_session=db_session
        )

        # Run workflow (async generator for event handling)
        async for event in workflow.run_async(ctx):

            # Check for HIL event
            if isinstance(event, HumanConsultationRequiredEvent):
                logger.info(f"HIL triggered for job {job_id}: {event.reason}")

                # Pause workflow - update job status
                job.status = JobStatus.AWAITING_APPROVAL
                job.requires_approval = True
                job.approval_reason = event.reason
                job.approval_timeout_at = datetime.utcnow() + timedelta(hours=1)  # 1-hour timeout

                # Store categorization details for frontend
                job.context["categorization_result"] = {
                    "category": event.current_value,
                    "confidence_score": event.confidence,
                    "reasoning": event.context.get("reasoning", ""),
                    "has_ambiguity_signals": event.context.get("has_ambiguity_signals", True),
                    "ambiguity_details": event.context.get("ambiguity_details", ""),
                    "alternative_categories": event.alternative_values
                }

                db_session.commit()

                # Wait for HumanResponseEvent (implementation-specific)
                human_response = await wait_for_human_response(job_id, timeout_seconds=3600)

                if not human_response:
                    # Timeout - reject job
                    job.status = JobStatus.REJECTED
                    job.context["rejection_reason"] = "Approval timeout (no human response within 1 hour)"
                    db_session.commit()
                    logger.warning(f"Job {job_id} rejected due to approval timeout")
                    return

                # Resume workflow with human decision
                logger.info(f"Resuming workflow for job {job_id} with human decision: {human_response.decision}")
                workflow.send_event(human_response)

            # Handle workflow completion
            elif isinstance(event, StopEvent):
                job.status = JobStatus.COMPLETED
                job.context["result"] = event.result
                db_session.commit()
                logger.info(f"Job {job_id} completed successfully")
                return

    except Exception as e:
        # Workflow failed
        job.status = JobStatus.FAILED
        job.context["error"] = str(e)
        job.context["traceback"] = traceback.format_exc()
        db_session.commit()
        logger.error(f"Job {job_id} failed: {e}")
        raise

async def wait_for_human_response(job_id: str, timeout_seconds: int = 3600) -> Optional[HumanResponseEvent]:
    """
    Wait for human approval decision.

    Implementation options:
    1. Poll database for approval_decisions table (simple, works with existing setup)
    2. Redis pub/sub (requires Redis)
    3. Workflow event queue (depends on workflow architecture)

    Returns HumanResponseEvent if approved, None if timeout.
    """

    # Option 1: Poll database (simplest for now)
    start_time = datetime.utcnow()
    poll_interval = 2  # Check every 2 seconds

    while (datetime.utcnow() - start_time).total_seconds() < timeout_seconds:
        # Check if approval decision exists
        with get_db_session() as db:
            job = db.query(Job).filter_by(id=job_id).first()

            if job.status == JobStatus.APPROVED:
                # Fetch approval record
                approval = db.query(ApprovalDecisionRecord).filter_by(job_id=job_id).order_by(
                    ApprovalDecisionRecord.created_at.desc()
                ).first()

                if approval:
                    return HumanResponseEvent(
                        consultation_type="gamp5_categorization",
                        decision=approval.human_category,
                        justification=approval.justification,
                        metadata={
                            "user_id": approval.user_id,
                            "digital_signature": approval.digital_signature
                        }
                    )

            elif job.status == JobStatus.REJECTED:
                # Explicitly rejected by human
                return None

        # Sleep before next poll
        await asyncio.sleep(poll_interval)

    # Timeout
    return None

## Risks

### Risk 1: Workflow Resumption Complexity
**Risk**: Resuming paused workflows may be complex depending on LlamaIndex architecture
**Mitigation**: Use polling-based approach (simple, reliable), consider Redis pub/sub for production
**Acceptance**: Workflow resumes within 5 seconds of approval submission

### Risk 2: Database Performance (Polling)
**Risk**: Worker polling database every 2 seconds creates load
**Mitigation**: Add database indexes, use connection pooling, limit poll duration to timeout
**Acceptance**: <10ms query time for approval check

### Risk 3: Timeout Edge Cases
**Risk**: User submits approval exactly at timeout boundary
**Mitigation**: Check timeout before processing approval, clear error message
**Acceptance**: No race conditions, clear rejection reason

**Key Patterns from Examples:**

1. **Event-based Pause** (from 5_human_in_the_loop.py):
```python
@step
async def step_requiring_input(self, ctx: Context, ev: SomeEvent) -> InputRequiredEvent | NextEvent:
    if needs_human_input:
        ctx.write_event_to_stream(InputRequiredEvent(
            prefix="Awaiting human decision",
            current_value=ai_recommendation
        ))
        return  # Workflow pauses here
    return NextEvent(result=result)

@step
async def handle_human_response(self, ctx: Context, ev: HumanResponseEvent) -> NextEvent:
    # Resume with human decision
    human_decision = ev.response
    return NextEvent(result=human_decision)
```

2. **External Event Injection** (for API integration):
```python
# In API endpoint
workflow_handler = workflow.run(ctx=context)

# Workflow pauses at InputRequiredEvent
# API stores job state as "awaiting_approval"

# Later, when human submits approval:
await workflow_handler.send_event(HumanResponseEvent(
    response=human_decision
))
```
