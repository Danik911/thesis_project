### Phase 1: Fix Langfuse @observe Decorator (1-2 hours)

**File:** `main/api/app.py`

**Problem:** `@observe` decorator serializes file uploads, causing hangs.

**Solution:** Use manual `langfuse.trace()` context manager with explicit input filtering.

```python
from langfuse import Langfuse

# Initialize Langfuse client (singleton)
langfuse_client = Langfuse()

@app.post("/jobs", response_model=JobSubmitResponse, status_code=status.HTTP_201_CREATED)
async def submit_job(
    file: ValidatedFileDep,
    current_user: CurrentUserDep,
    job_queue: JobQueueDep,
    job_repository: JobRepositoryDep,
    job_lock: JobLockDep,
    storage_adapter: StorageAdapterDep,
    db_job_repository: DbJobRepositoryDep
) -> JobSubmitResponse:
    """Submit URS for test generation with manual Langfuse tracing."""

    job_id = str(uuid.uuid4())

    # Manual tracing - explicitly control what gets logged
    trace = langfuse_client.trace(
        name="create_test_generation_job",
        user_id=current_user.sub,
        metadata={
            "job_id": job_id,
            "filename": file.filename,
            "content_length": len(await file.read()),  # Safe - just the length
            "endpoint": "POST /jobs"
        }
    )

    try:
        # Reset file position after reading length
        await file.seek(0)

        # ... rest of job submission logic ...

        trace.update(output={"job_id": job_id, "status": "submitted"})
        return JobSubmitResponse(job_id=job_id, status="pending")

    except Exception as e:
        trace.update(
            output={"error": str(e)},
            level="ERROR"
        )
        raise
    finally:
        # Ensure trace is flushed even on error
        langfuse_client.flush()
```

**Alternative:** Create a utility decorator that filters inputs:

```python
# main/api/observability.py (NEW FILE)
from functools import wraps
from langfuse import Langfuse
from fastapi import UploadFile

def observe_safe(name: str, exclude_params: list[str] = None):
    """
    Safe @observe decorator that excludes non-serializable params.

    Usage:
        @observe_safe(name="submit_job", exclude_params=["file"])
        async def submit_job(file: UploadFile, ...):
    """
    exclude_params = exclude_params or []
    langfuse = Langfuse()

    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # Filter out excluded params
            safe_kwargs = {
                k: (f"<{type(v).__name__}>" if k in exclude_params else v)
                for k, v in kwargs.items()
            }

            trace = langfuse.trace(name=name, input=safe_kwargs)
            try:
                result = await func(*args, **kwargs)
                trace.update(output=result.dict() if hasattr(result, 'dict') else str(result))
                return result
            except Exception as e:
                trace.update(output={"error": str(e)}, level="ERROR")
                raise
            finally:
                langfuse.flush()
        return wrapper
    return decorator
```

```python
# main/api/worker.py

async def process_approved_jobs(db_repository: PostgresJobRepository):
    """
    Poll for APPROVED jobs and resume their workflows.

    Called periodically by worker main loop.
    """
    approved_jobs = await db_repository.get_jobs_by_status("APPROVED")

    for job in approved_jobs:
        job_id = job["id"]
        logger.info(f"[WORKER] Resuming approved job: {job_id}")

        try:
            # 1. Transition job to PROCESSING
            await db_repository.update_job_status(job_id, "PROCESSING")

            # 2. Fetch approval decision
            approval_record = await db_repository.get_latest_approval(job_id)
            if not approval_record:
                logger.error(f"[WORKER] No approval record found for job {job_id}")
                await db_repository.update_job_status(job_id, "FAILED")
                continue

            # 3. Create HumanResponseEvent with approved category
            human_response = HumanResponseEvent(
                consultation_type="gamp5_categorization",
                decision=approval_record["human_category"],
                justification=approval_record["justification"],
                metadata={
                    "user_id": approval_record["user_id"],
                    "digital_signature": approval_record["digital_signature"],
                    "approved_at": approval_record["timestamp"].isoformat()
                }
            )

            # 4. Load workflow context from job
            job_context = job.get("context", {})
            urs_content = job.get("urs_content") or job_context.get("urs_content")

            if not urs_content:
                logger.error(f"[WORKER] No URS content found for job {job_id}")
                await db_repository.update_job_status(job_id, "FAILED")
                continue

            # 5. Re-run workflow with human response injected
            result = await execute_workflow_with_approval(
                job_id=job_id,
                urs_content=urs_content,
                human_response=human_response,
                db_repository=db_repository
            )

            # 6. Update job with result
            await db_repository.update_job_status(job_id, "COMPLETED")
            await db_repository.update_job_context(job_id, {"result": result})

            logger.info(f"[WORKER] Job {job_id} completed successfully after approval")

        except Exception as e:
            logger.exception(f"[WORKER] Failed to resume job {job_id}: {e}")
            await db_repository.update_job_status(job_id, "FAILED")
            await db_repository.update_job_context(job_id, {"error": str(e)})


async def execute_workflow_with_approval(
    job_id: str,
    urs_content: str,
    human_response: HumanResponseEvent,
    db_repository: PostgresJobRepository
) -> dict:
    """
    Execute workflow with pre-approved human decision.

    Instead of running full workflow and waiting for approval,
    this runs the workflow with the human decision already injected.

    Two implementation options:

    Option A: Skip categorization step (use approved category directly)
    Option B: Re-run workflow with human response pre-loaded
    """

    # Option A: Skip categorization, use approved category
    # This is simpler and more efficient

    from main.api.worker_executor import WorkflowExecutor

    executor = WorkflowExecutor()

    # Create workflow context with pre-approved category
    context = {
        "job_id": job_id,
        "urs_content": urs_content,
        "pre_approved_category": human_response.decision,
        "approval_metadata": human_response.metadata,
        "skip_categorization": True  # Signal to skip HIL step
    }

    # Run workflow (categorization step will see skip_categorization=True and use pre_approved_category)
    result = await executor.run_workflow(context)

    return result
```

```python
# main/api/worker_executor.py

async def run_workflow(self, context: dict) -> dict:
    """Run the unified workflow."""

    # Check for pre-approved category (workflow resumption after HIL)
    if context.get("skip_categorization") and context.get("pre_approved_category"):
        logger.info(f"[EXECUTOR] Using pre-approved category: {context['pre_approved_category']}")

        # Inject category directly into workflow, skipping categorization step
        categorization_result = CategorizationResult(
            gamp_category=context["pre_approved_category"],
            confidence_score=1.0,  # Human approval = 100% confidence
            reasoning="Human-approved categorization",
            requires_human_review=False,  # Already reviewed
            approval_metadata=context.get("approval_metadata", {})
        )

        # Run workflow from test generation step (skip categorization)
        result = await self._run_from_test_generation(
            urs_content=context["urs_content"],
            categorization_result=categorization_result
        )

        return result

    # Normal flow - run full workflow
    return await self._run_full_workflow(context)
```
