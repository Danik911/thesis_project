### 1. Create RBAC FastAPI Dependency

Integrate the existing RBAC system with FastAPI:

```python
# main/api/dependencies.py

from fastapi import Depends, HTTPException, status, Request
from src.compliance.rbac_system import (
    get_rbac_system,
    RoleBasedAccessControl,
    Permission,
    PharmaceuticalRole,
    UserSession
)

class RBACDependency:
    """
    FastAPI dependency for RBAC permission checks.

    Implements 21 CFR Part 11 §11.10(d): Limiting system access to authorized individuals.
    """

    def __init__(self, required_permission: Permission):
        self.required_permission = required_permission
        self.rbac = get_rbac_system()

    async def __call__(
        self,
        request: Request,
        user: ClerkClaims = Depends(get_current_user),
    ) -> UserSession:
        """
        Verify user has required permission.

        Flow:
        1. Get or create RBAC session from Clerk user
        2. Verify session is valid (not expired, not locked)
        3. Check permission for the required action
        4. Log access attempt for audit trail
        """
        # Map Clerk user to RBAC session
        session = await self._get_or_create_session(user, request)

        if not session or not session.is_valid():
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Session expired or invalid. Please re-authenticate."
            )

        # Check permission
        has_permission = self.rbac.check_permission(
            session_id=session.session_id,
            permission=self.required_permission,
            resource_context={"endpoint": str(request.url.path)}
        )

        if not has_permission:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Permission denied: {self.required_permission.value} required"
            )

        return session

    async def _get_or_create_session(
        self,
        user: ClerkClaims,
        request: Request
    ) -> UserSession | None:
        """
        Get existing session or create new one from Clerk claims.

        Maps Clerk user to pharmaceutical role based on Clerk metadata.
        """
        # Check for existing session
        session_id = request.headers.get("X-Session-ID")
        if session_id:
            session = self.rbac.active_sessions.get(session_id)
            if session and session.is_valid():
                session.update_activity()
                return session

        # Create new session from Clerk user
        role = await self._get_user_role(user)
        if not role:
            return None

        device_info = {
            "device_id": request.headers.get("X-Device-ID", "unknown"),
            "ip_address": request.client.host if request.client else "unknown",
            "user_agent": request.headers.get("User-Agent", "unknown")
        }

        return self.rbac.authenticate_user(
            user_id=user.sub,
            device_info=device_info
        )

    async def _get_user_role(self, user: ClerkClaims) -> PharmaceuticalRole | None:
        """
        Get pharmaceutical role from Clerk user metadata.

        In production, this would read from Clerk's publicMetadata or privateMetadata.
        Default to TEST_GENERATOR_USER for basic access.
        """
        # TODO: Read role from Clerk metadata
        # For now, check if user exists in RBAC registry
        if user.sub in self.rbac.user_registry:
            role_str = self.rbac.user_registry[user.sub].get("role")
            return PharmaceuticalRole(role_str) if role_str else None

        # Default role for new users (minimal permissions)
        return PharmaceuticalRole.TEST_GENERATOR_USER


# Permission-specific dependencies
def require_permission(permission: Permission):
    """Factory for permission-specific dependencies."""
    return Depends(RBACDependency(permission))

# Common permission aliases
RequireCreateTests = require_permission(Permission.CREATE_TESTS)
RequireApproveTests = require_permission(Permission.APPROVE_TESTS)
RequireAccessAuditTrail = require_permission(Permission.ACCESS_AUDIT_TRAIL)
RequireManageUsers = require_permission(Permission.MANAGE_USERS)
```

### 3. Clerk Role Synchronization

Sync pharmaceutical roles with Clerk user metadata:

```python
# main/api/clerk_rbac_sync.py

from clerk_backend_api import Clerk
from src.compliance.rbac_system import (
    get_rbac_system,
    PharmaceuticalRole
)

class ClerkRBACSync:
    """
    Synchronize Clerk users with RBAC system.

    Reads role from Clerk publicMetadata.pharmaceutical_role
    and registers/updates in RBAC system.
    """

    def __init__(self):
        self.clerk = Clerk(api_key=os.getenv("CLERK_SECRET_KEY"))
        self.rbac = get_rbac_system()

    async def sync_user(self, clerk_user_id: str) -> PharmaceuticalRole:
        """
        Sync single user from Clerk to RBAC.

        Reads role from Clerk metadata, registers in RBAC if not exists.
        """
        user = await self.clerk.users.get(clerk_user_id)

        # Get role from Clerk metadata
        role_str = user.public_metadata.get(
            "pharmaceutical_role",
            "test_generator_user"  # Default role
        )
        role = PharmaceuticalRole(role_str)

        # Register or update in RBAC
        if clerk_user_id not in self.rbac.user_registry:
            self.rbac.register_user(
                user_id=clerk_user_id,
                user_name=f"{user.first_name} {user.last_name}",
                role=role,
                contact_info={"email": user.email_addresses[0].email_address},
                training_completed=user.public_metadata.get("training_completed", False)
            )

        return role

    async def set_user_role(
        self,
        clerk_user_id: str,
        new_role: PharmaceuticalRole
    ) -> None:
        """
        Update user's pharmaceutical role in Clerk metadata.

        Called by admin users to assign roles.
        """
        await self.clerk.users.update(
            clerk_user_id,
            public_metadata={
                "pharmaceutical_role": new_role.value
            }
        )

        # Update RBAC registry
        if clerk_user_id in self.rbac.user_registry:
            self.rbac.user_registry[clerk_user_id]["role"] = new_role.value
            self.rbac._save_user_registry()
```

## Best Practices

1. **Separation of Duties**: QA cannot approve their own test submissions
2. **Least Privilege**: Default to TEST_GENERATOR_USER with minimal permissions
3. **Audit Everything**: Log all permission checks (already in RBAC system)
4. **Session Management**: Respect session timeout (8 hours default)
5. **Fail Closed**: If RBAC check fails, deny access explicitly

## Common Issues to Avoid

1. **Forgetting session cleanup**: Call `cleanup_expired_sessions()` periodically
2. **Hardcoding roles**: Always read from Clerk metadata, not code
3. **Missing training check**: RBAC requires `training_completed=True`
4. **Ignoring device checks**: Device info required for authentication
5. **Not syncing with Clerk**: Role changes in Clerk must update RBAC
