**SQL Schema**:

```sql
-- Workflow steps table for real-time progress tracking
CREATE TABLE workflow_steps (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    job_id UUID NOT NULL REFERENCES jobs(id) ON DELETE CASCADE,
    step_id INTEGER NOT NULL,              -- 1-10 (workflow step number)
    step_name VARCHAR(100) NOT NULL,       -- e.g., "GAMP-5 Categorization"
    status VARCHAR(20) NOT NULL,           -- pending, active, completed, failed
    compliance_tags TEXT[],                -- e.g., {"GAMP-5", "ALCOA+"}
    started_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,
    error_message TEXT,                    -- NULL unless status=failed
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_job FOREIGN KEY (job_id) REFERENCES jobs(id) ON DELETE CASCADE,
    CONSTRAINT unique_job_step UNIQUE (job_id, step_id),
    CONSTRAINT valid_status CHECK (status IN ('pending', 'active', 'completed', 'failed'))
);

-- Index for fast lookups by job_id
CREATE INDEX idx_workflow_steps_job_id ON workflow_steps(job_id);

-- Index for status queries
CREATE INDEX idx_workflow_steps_status ON workflow_steps(status);
```

```python
from main.api.models import WorkflowStep, WorkflowStepStatus
from main.api.workflow_step_definitions import get_workflow_step_definition
from datetime import datetime

def track_workflow_step(step_id: int):
    """
    Decorator to track workflow step status in database.
    Wraps existing @step decorators.
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(ctx: Context, ev: Event) -> Event:
            job_id = ctx.data.get("job_id")
            db_session = ctx.data.get("db_session")

            # Update step status to 'active'
            step = db_session.query(WorkflowStep).filter_by(
                job_id=job_id,
                step_id=step_id
            ).first()

            if step:
                step.status = WorkflowStepStatus.ACTIVE
                step.started_at = datetime.utcnow()
                db_session.commit()

            try:
                # Execute original workflow step function
                result = await func(ctx, ev)

                # Update step status to 'completed'
                if step:
                    step.status = WorkflowStepStatus.COMPLETED
                    step.completed_at = datetime.utcnow()
                    db_session.commit()

                return result

            except Exception as e:
                # Update step status to 'failed'
                if step:
                    step.status = WorkflowStepStatus.FAILED
                    step.completed_at = datetime.utcnow()
                    step.error_message = str(e)
                    db_session.commit()

                # Re-raise exception (NO FALLBACK LOGIC)
                raise

        return wrapper
    return decorator
```

```python
from main.api.models import WorkflowStep, WorkflowStepResponse

@router.get("/jobs/{job_id}/workflow-steps", response_model=List[WorkflowStepResponse])
async def get_workflow_steps(
    job_id: str,
    claims: ClerkClaims = Depends(verify_token),
    db: Session = Depends(get_db)
):
    """
    Get workflow step status for a job.

    Returns array of 10 workflow steps with real-time status:
    - pending: Not yet started
    - active: Currently executing
    - completed: Successfully finished
    - failed: Encountered error

    Polling Interval: Frontend should poll every 2 seconds during job execution.
    """
    # Verify job ownership
    job = db.query(Job).filter_by(id=job_id).first()
    if not job:
        raise HTTPException(404, "Job not found")

    if job.user_id != claims.user_id:
        raise HTTPException(403, "Not authorized to access this job")

    # Fetch workflow steps ordered by step_id
    steps = db.query(WorkflowStep).filter_by(job_id=job_id).order_by(WorkflowStep.step_id).all()

    if not steps:
        # Job created but workflow not started yet → initialize steps
        from main.api.workflow_step_definitions import initialize_workflow_steps
        steps = initialize_workflow_steps(job_id, db)

    return [WorkflowStepResponse.from_orm(step) for step in steps]
```

```tsx
import { useState, useEffect } from 'react';
import WorkflowStepCard from './WorkflowStepCard';
import { getCardById, ComplianceCard } from '@/lib/complianceContent';

interface WorkflowStep {
  step_id: number;
  step_name: string;
  status: 'pending' | 'active' | 'completed' | 'failed';
  compliance_tags: string[];
  error_message?: string;
}

interface WorkflowFlashCardsProps {
  jobId: string;
}

// Map workflow steps to flash card IDs
const STEP_TO_CARD_MAPPING: Record<number, string | null> = {
  1: 'iso-malware',                      // OWASP LLM Top 10
  2: 'gamp5-categorization',             // GAMP-5 Software Categorization
  3: 'q9-risk-assessment',               // ICH Q9 Risk Assessment
  4: 'euai-accuracy',                    // EU AI Act Multi-Agent Validation
  5: 'euai-data-governance',             // EU AI Act Data Governance
  6: 'euai-human-oversight',             // 21 CFR Part 11 + EU AI Act Human Oversight
  7: 'cfr11-signatures',                 // 21 CFR Part 11 Electronic Signatures
  8: 'gamp5-validation-testing',         // GAMP-5 Validation Testing (OQ)
  9: 'alcoa-accurate',                   // ALCOA+ (could cycle through all 9)
  10: 'cfr11-audit-trail'                // 21 CFR Part 11 Audit Trail
};

export default function WorkflowFlashCards({ jobId }: WorkflowFlashCardsProps) {
  const [steps, setSteps] = useState<WorkflowStep[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  // Fetch workflow steps every 2 seconds
  useEffect(() => {
    const fetchSteps = async () => {
      try {
        const response = await fetch(`/api/jobs/${jobId}/workflow-steps`, {
          headers: {
            Authorization: `Bearer ${await getToken()}`
          }
        });

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }

        const data = await response.json();
        setSteps(data);
        setLoading(false);
      } catch (err) {
        console.error('Error fetching workflow steps:', err);
        setError(err.message);
        setLoading(false);
      }
    };

    // Initial fetch
    fetchSteps();

    // Poll every 2 seconds
    const interval = setInterval(fetchSteps, 2000);

    // Cleanup
    return () => clearInterval(interval);
  }, [jobId]);

  if (loading) {
    return (
      <div className="flex items-center justify-center p-8">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600" />
        <span className="ml-3 text-gray-600">Loading workflow steps...</span>
      </div>
    );
  }

  if (error) {
    return (
      <div className="p-4 bg-red-50 border border-red-300 rounded text-red-800">
        <strong>Error loading workflow steps:</strong> {error}
      </div>
    );
  }

  return (
    <div className="workflow-flash-cards space-y-4">
      <h2 className="text-xl font-bold text-gray-900">
        Workflow Progress
      </h2>

      {/* Progress Bar */}
      <div className="relative h-2 bg-gray-200 rounded-full overflow-hidden">
        <div
          className="absolute h-full bg-blue-600 transition-all duration-500"
          style={{
            width: `${(steps.filter(s => s.status === 'completed').length / steps.length) * 100}%`
          }}
        />
      </div>

      {/* Step Cards Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {steps.map(step => {
          // Get flash card for this step
          const cardId = STEP_TO_CARD_MAPPING[step.step_id];
          const card = cardId ? getCardById(cardId) : undefined;

          return (
            <WorkflowStepCard
              key={step.step_id}
              stepNumber={step.step_id}
              stepName={step.step_name}
              status={step.status}
              complianceTags={step.compliance_tags}
              card={card}
              errorMessage={step.error_message}
            />
          );
        })}
      </div>

      {/* Active Step Highlight */}
      {steps.some(s => s.status === 'active') && (
        <div className="mt-4 p-4 bg-blue-50 border border-blue-300 rounded">
          <p className="text-sm text-blue-800">
            <strong>Currently Running:</strong>{' '}
            {steps.find(s => s.status === 'active')?.step_name}
          </p>
        </div>
      )}
    </div>
  );
}
```

**Features**:
- Real-time polling (every 2 seconds)
- Progress bar shows completion percentage
- Active step highlighted with glow effect
- Flash cards appear for active step only (educational)
- Responsive grid layout (1 column mobile, 2 tablet, 3 desktop)

## Risks

### Risk 1: Database Performance
**Risk**: Polling `/workflow-steps` every 2 seconds creates database load
**Mitigation**: Add database index on `job_id`, use connection pooling, cache responses for 1 second
**Acceptance**: <50ms query time for workflow step lookup

### Risk 2: Workflow Step Tracking Overhead
**Risk**: Decorator adds latency to each workflow step
**Mitigation**: Use async database updates, batch status changes
**Acceptance**: <100ms overhead per step

### Risk 3: Flash Card Mismatch
**Risk**: Flash card content doesn't match actual workflow step
**Mitigation**: Validate `STEP_TO_CARD_MAPPING` against `complianceContent.ts` card IDs
**Acceptance**: All 10 steps have valid flash card mapping

### Risk 4: Real-Time Update Lag
**Risk**: Frontend shows outdated step status due to polling delay
**Mitigation**: Use 2-second polling interval (acceptable for 8-12 minute workflows)
**Acceptance**: Maximum 2-second lag between step completion and UI update
